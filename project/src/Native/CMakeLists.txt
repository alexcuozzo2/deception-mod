cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(DeceptionNativeBridges VERSION 1.0 LANGUAGES CXX)

# Building an own-helper binary never admits a game/platform runtime profile.
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Windows|Darwin|Linux)$")
  message(FATAL_ERROR "Only Windows, Darwin and Linux build recipes exist")
endif()
if(CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "This tested-pack recipe requires target-host execution; cross compilation is not admission")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Deception native bridges require 64-bit pointers")
endif()
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#if !defined(__x86_64__) && !defined(_M_X64)\n#error Target must be x64\n#endif\nint main(){return 0;}" DCP_TARGET_X64)
if(NOT DCP_TARGET_X64)
  message(FATAL_ERROR "Deception native bridges require the x86-64 target, not merely a 64-bit host")
endif()
if(APPLE AND NOT CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64")
  message(FATAL_ERROR "Set CMAKE_OSX_ARCHITECTURES=x86_64 explicitly; no universal or ARM artifact")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(MSVC|GNU|Clang|AppleClang)$")
  message(FATAL_ERROR "Unreviewed C++ compiler")
endif()
if(WIN32 AND NOT MSVC)
  message(FATAL_ERROR "Windows requires the MSVC ABI toolchain; MinGW exception compatibility is unverified")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS OFF)
if(MSVC)
  # /EHc assumes extern-C entries cannot throw. These entries intentionally
  # propagate native C++ exceptions: remove CMake's default /EHsc, then use
  # synchronous C++ unwind with extern-C exceptions allowed on BOTH sides.
  foreach(flags CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE CMAKE_CXX_FLAGS_RELWITHDEBINFO CMAKE_CXX_FLAGS_MINSIZEREL)
    string(REGEX REPLACE "[/\\-]EH[a-zA-Z0-9-]+" "" ${flags} "${${flags}}")
  endforeach()
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
find_package(Threads REQUIRED)
include(CTest)
if(NOT BUILD_TESTING)
  message(FATAL_ERROR "This recipe always builds and runs the three native harnesses")
endif()

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/bridge-build-contract.json" contract)
set(index "{\"schema\":1,\"system\":\"${CMAKE_SYSTEM_NAME}\",\"compiler\":\"${CMAKE_CXX_COMPILER_ID}\",\"compilerVersion\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"pointerSize\":${CMAKE_SIZEOF_VOID_P},\"configuration\":\"$<CONFIG>\",\"bridges\":{")
set(separator "")
foreach(kind state execution command)
  set(lib "deception-${kind}-bridge")
  set(harness "${kind}-bridge-harness")
  add_library(${lib} SHARED "deception-${kind}-bridge.cpp")
  add_executable(${harness} "../../tests/native/${kind}-bridge-harness.cpp")
  target_compile_definitions(${harness} PRIVATE DCP_BRIDGE_IMPORT)
  target_link_libraries(${harness} PRIVATE ${lib})
  foreach(target ${lib} ${harness})
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(MSVC)
      target_compile_options(${target} PRIVATE /EHs /EHc- /W4 /WX /permissive-)
      target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    else()
      target_compile_options(${target} PRIVATE -fexceptions -Wall -Wextra -Werror -pedantic)
    endif()
    set_target_properties(${target} PROPERTIES
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>"
      LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>"
      ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/$<CONFIG>")
  endforeach()
  # Restrict exported implementation symbols, including weak STL instantiations.
  string(JSON prefix GET "${contract}" bridges ${kind} prefix)
  string(JSON length LENGTH "${contract}" bridges ${kind} exports)
  math(EXPR last "${length} - 1")
  set(exports "")
  foreach(i RANGE 0 ${last})
    string(JSON suffix GET "${contract}" bridges ${kind} exports ${i})
    if(APPLE)
      string(APPEND exports "_${prefix}${suffix}\n")
    else()
      string(APPEND exports "    ${prefix}${suffix};\n")
    endif()
  endforeach()
  if(APPLE)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${kind}.exports" "${exports}")
    target_link_options(${lib} PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_BINARY_DIR}/${kind}.exports")
    set_target_properties(${lib} PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
    set_target_properties(${harness} PROPERTIES BUILD_RPATH "@loader_path")
  elseif(UNIX)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${kind}.exports" "{\n  global:\n${exports}  local: *;\n};\n")
    target_link_options(${lib} PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_BINARY_DIR}/${kind}.exports" "LINKER:-z,defs")
    set_target_properties(${harness} PROPERTIES BUILD_RPATH "$ORIGIN" BUILD_RPATH_USE_ORIGIN ON)
  endif()
  add_test(NAME "${kind}-bridge" COMMAND ${harness})
  set_tests_properties("${kind}-bridge" PROPERTIES TIMEOUT 90 LABELS "native-boundary;mock-only")
  string(APPEND index "${separator}\"${kind}\":{\"library\":\"$<TARGET_FILE:${lib}>\",\"harness\":\"$<TARGET_FILE:${harness}>\"}")
  set(separator ",")
endforeach()
string(APPEND index "}}\n")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/artifact-index-$<CONFIG>.json" CONTENT "${index}")
